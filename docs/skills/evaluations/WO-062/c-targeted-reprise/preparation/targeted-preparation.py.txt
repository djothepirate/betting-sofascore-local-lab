"""Targeted preparation: reuses unchanged frozen inputs and evaluator for JM-N01."""
import importlib.util,json,pathlib,sys,copy
ROOT=pathlib.Path(__file__).resolve().parent.parent
spec=importlib.util.spec_from_file_location('c_previous',ROOT/'.tmp/wo062_evaluate_c.py')
m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)
original_read=m.read
skill,case_id=sys.argv[2:4]
assert (skill,case_id) in [('ss-java-module','JM-N01'),('ss-windows-runtime','WR-H01')]
def selected_read(path):
    value=original_read(path)
    if pathlib.Path(path).name=='cases.json':
        value=copy.deepcopy(value);value['cases']=[c for c in value['cases'] if c['id']==case_id]
        assert len(value['cases'])==1;value['case_count']=1
    return value
m.read=selected_read
if sys.argv[1]=='prepare':m.prepare(skill,'run-03')
elif sys.argv[1]=='run':
    assert case_id=='JM-N01','WR-H01 requires timed evaluator and prepared flow addendum.'
    raise SystemExit(m.run_case(skill,'run-03',case_id))
else:raise ValueError('Unknown mode')
