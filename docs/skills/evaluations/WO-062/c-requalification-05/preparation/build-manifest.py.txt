"""Create the non-self-referential manifest for immutable C5 preparation artifacts."""
import hashlib
import json
import pathlib
import subprocess
from datetime import datetime, timezone

ROOT = pathlib.Path(__file__).resolve().parent.parent
PREPARATION = ROOT / 'docs' / 'skills' / 'evaluations' / 'WO-062' / 'c-requalification-05' / 'preparation'
manifest = PREPARATION / 'artifact-manifest.json'
assert not manifest.exists(), manifest
files = []
for path in sorted(PREPARATION.rglob('*')):
    if not path.is_file():
        continue
    files.append({
        'path': path.relative_to(PREPARATION).as_posix(),
        'bytes': path.stat().st_size,
        'sha256': hashlib.sha256(path.read_bytes()).hexdigest(),
    })
result = {
    'schema': 'wo062-c5-preparation-artifact-manifest-v1',
    'created_at_utc': datetime.now(timezone.utc).isoformat(),
    'source_head': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip(),
    'files': files,
    'note': 'The manifest does not hash itself. It is created before the model sessions and records only C5 preparation, not their results.',
}
manifest.write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n', encoding='utf-8', newline='\n')
print(json.dumps({'files': len(files), 'manifest': str(manifest)}))
