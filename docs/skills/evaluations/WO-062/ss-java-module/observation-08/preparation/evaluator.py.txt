def write(p,v):p.write_text(json.dumps(v,ensure_ascii=False,indent=2)+'\n',encoding='utf-8',newline='\n')
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def strings(v):
    if isinstance(v,str):yield v
    elif isinstance(v,list):
        for x in v:yield from strings(x)
    elif isinstance(v,dict):
        for x in v.values():yield from strings(x)

def runtime_execution_for(skill,case):
    mode=case.get('execution_mode','read_only')
    if mode not in ('read_only','bounded_local_runtime'):
        raise ValueError('Unsupported execution_mode')
    runtime_execution=mode=='bounded_local_runtime'
    caseid=case.get('case_id',case.get('id'))
    if runtime_execution and not (skill=='ss-windows-runtime' and caseid in ('WR-H01','WR-N01') and not case['kind'].startswith('selection')):
        raise ValueError('workspace-write is restricted to non-selection WR-H01/WR-N01 of ss-windows-runtime')
    return runtime_execution

def command_text(item):
    return str(item.get('command',''))

def read_policy_audit(caseid,commands,case):
    """Record C5 reading signals from completed commands without changing the evidence."""
    audit={'schema':'wo062-c5-read-policy-audit-v1','case_id':caseid,
           'scope':'Completed command_execution events only; an aid to independent review, not a sandbox barrier.',
           'runtime_dependency_direct_access':[], 'repeated_readable_access':[],
           'readable_commands':[], 'status':'PASS'}
    if caseid!='WR-H01':
        audit['status']='NOT_APPLICABLE'
        return audit
    runtime_only=[str(x).replace('\\','/').lower() for x in case.get('runtime_dependency_only',[])]
    readable=[str(x).replace('\\','/').lower() for x in case.get('readable_input_paths',[])]
    text_read_markers=('get-content','cat ','type ','readalltext','select-string','rg ','findstr','import-module','copy-item','get-filehash')
    seen={p:[] for p in readable}
    for idx,item in enumerate(commands,1):
        raw=command_text(item)
        normalized=raw.replace('\\','/').lower()
        if any(marker in normalized for marker in text_read_markers):
            for dependency in runtime_only:
                if dependency in normalized:
                    audit['runtime_dependency_direct_access'].append({'command_index':idx,'path':dependency,'command':raw[:1200]})
            for path in readable:
                if path in normalized:
                    seen[path].append({'command_index':idx,'command':raw[:1200]})
    for path,accesses in seen.items():
        if accesses:
            audit['readable_commands'].append({'path':path,'access_count':len(accesses),'accesses':accesses})
        if len(accesses)>1:
            audit['repeated_readable_access'].append({'path':path,'accesses':accesses})
    if audit['runtime_dependency_direct_access'] or audit['repeated_readable_access']:
        audit['status']='FAIL'
    return audit

def run_timed_capture(args,stdin,stdout,stderr,timeout,out):
    import threading
    started=time.monotonic();started_utc=utc();errors=[];count=[0]
    process=subprocess.Popen(args,stdin=stdin,stdout=subprocess.PIPE,stderr=stderr)
    def consume():
        try:
            with (out/'native-timing.jsonl').open('xb') as timing:
                for line in iter(process.stdout.readline,b''):
                    stdout.write(line);stdout.flush()
                    meta={'sequence':count[0],'received_at_utc':utc(),'elapsed_seconds':round(time.monotonic()-started,6),'raw_line_sha256':hashlib.sha256(line).hexdigest(),'bytes':len(line)}
                    count[0]+=1
                    try:
                        event=json.loads(line);item=event.get('item',{})
                        meta.update({'event_type':event.get('type'),'item_id':item.get('id'),'item_type':item.get('type')})
                        if item.get('type')=='command_execution':
                            output=item.get('aggregated_output','')
                            meta.update({'command_chars':len(item.get('command','')),'output_chars':len(output),'exit_code':item.get('exit_code')})
                            meta['phase_markers']=re.findall(r'WR_PHASE=([A-Z_]+)',output)
                    except (ValueError,TypeError):meta['parse_error']=True
                    timing.write((json.dumps(meta,ensure_ascii=False)+'\n').encode());timing.flush()
        except Exception as exc:errors.append(type(exc).__name__)
    reader=threading.Thread(target=consume,daemon=True);reader.start();timed_out=False
    try:process.wait(timeout=timeout)
    except subprocess.TimeoutExpired:
        timed_out=True;process.kill();process.wait(timeout=5)
    reader.join(timeout=5)
    summary={'schema':'wo062-c5-event-timing-v1','started_at_utc':started_utc,'ended_at_utc':utc(),'duration_seconds':round(time.monotonic()-started,6),'owned_pid':process.pid,'exit_code':process.returncode,'timed_out':timed_out,'reader_finished':not reader.is_alive(),'reader_errors':errors,'event_count':count[0],'scope':'Local receipt times of CLI JSONL events; not server token timings or proof of model thinking duration.'}
    write(out/'native-timing-summary.json',summary)
    if not reader.is_alive():process.stdout.close()
    if timed_out:raise subprocess.TimeoutExpired(args,timeout)
    if reader.is_alive() or errors:raise RuntimeError('Timing/output reader incomplete')
    return subprocess.CompletedProcess(args,process.returncode)
