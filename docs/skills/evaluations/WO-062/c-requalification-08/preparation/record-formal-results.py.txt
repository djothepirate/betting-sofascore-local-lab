def sha256(path):
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()


def read_json(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))


def write_json(path, value):
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')


def review_markdown(case_id, verdict, summary, criteria, limitations):
    lines = [
        f'# Revue C8 — {case_id}',
        '',
        f'**Verdict : {verdict}.**',
        '',
        summary,
        '',
        '## Critères',
        '',
    ]
    for criterion in criteria:
        lines.extend([
            f"- **{criterion['criterion_id']} — {criterion['result']}** : {criterion['evidence']}",
        ])
    lines.extend(['', '## Limites', ''])
    lines.extend([f'- {limit}' for limit in limitations])
    lines.append('')
    return '\n'.join(lines)
