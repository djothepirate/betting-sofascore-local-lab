import pathlib,re,subprocess,json,hashlib
from datetime import datetime,timezone
root=pathlib.Path(__file__).resolve().parent.parent
base='6648dd423e556b5248b8793a539ae85a7680f9bc'
policy=(root/'ci/check-no-secrets.sh').read_text(encoding='utf-8')
patterns=[]
for line in policy.splitlines():
 m=re.fullmatch(r"([a-z_]+)_pattern='(.*)'",line)
 if m:patterns.append((m[1],re.compile(m[2].replace('[[:space:]]',r'\s'),re.IGNORECASE)))
assert len(patterns)==9
tracked=subprocess.check_output(['git','diff','--name-only','--diff-filter=AM',base,'--'],cwd=root,text=True).splitlines()
untracked=subprocess.check_output(['git','ls-files','--others','--exclude-standard'],cwd=root,text=True).splitlines()
paths=sorted(set(tracked+untracked));findings=[];files=[]
for rel in paths:
 p=root/rel
 if not p.is_file():continue
 b=p.read_bytes();txt=b.decode('utf-8')
 files.append({'path':rel,'sha256':hashlib.sha256(b).hexdigest()})
 for name,pattern in patterns:
  if pattern.search(txt):findings.append({'path':rel,'rule':name})
result={'checked_at_utc':datetime.now(timezone.utc).isoformat(),'status':'FAIL' if findings else 'PASS_HIGH_CONFIDENCE_DIFF','scope':'Current complete bytes of changed/added files since '+base,'method':'Nine current ci/check-no-secrets.sh regex families, evaluated locally with Python re; POSIX space translated to \\s; case insensitive. Not a CI invocation nor full-history scan. No matched value printed.','file_count':len(files),'findings':findings,'files':files}
(root/'.tmp/c-secret-check.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps({k:result[k] for k in ['status','file_count','findings']}))
raise SystemExit(bool(findings))
